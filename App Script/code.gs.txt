/**
 * Code.gs - Google Apps Script Backend for Process LOB Dashboard
 * Serves the HTML interface and provides data from the Google Sheet.
 */

/**
 * doGet - Entry point for the web app
 * Serves the main HTML page (index.html)
 */
function doGet() {
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle('Process LOB Dashboard')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

        };
        const mappedKey = columnMap[key] || key;
        obj[mappedKey] = row[index] !== undefined ? row[index].toString() : '';
      });
      
      // Derive Phase from MileStone if not present
      if (!obj.Phase) {
        const ms = obj.MileStone || '';
        if (ms.includes('Phase 1') || ms.includes('MilStone 1')) {
          obj.Phase = 'Phase 1';
        } else if (ms.includes('Phase 2') || ms.includes('MilStone 2')) {
          obj.Phase = 'Phase 2';
        } else {
          obj.Phase = 'Phase 1'; // Default fallback
        }
      }
      
      return obj;
    });
    
    return rows;
  } catch (error) {
    console.error('Error fetching sheet data:', error);
    // Return sample data as fallback
    return getSampleData();
  }
}
